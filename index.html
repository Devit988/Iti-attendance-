Administrator access

Admin access is granted only to the single Google account configured in the site code and in firestore.rules, and only when its email is verified. There is no PIN. Signing out removes admin access immediately.

Private trainee data (registration IDs, marks, attendance) is stored in the admin-only iti_private collection and is not readable by visitors.

Firestore Security Rules 
rules_version = '2';

// ITI CONNECT — Firestore Security Rules (deny by default)
//
// BEFORE DEPLOYING: replace ADMIN_EMAIL below with the SAME verified Google account you set in
//   CONFIGURED_ADMIN_EMAIL inside the website HTML (lower-case address, e.g. you@gmail.com).
//   While it still says ADMIN_EMAIL nobody is admin (fail-closed): the private data stays unreadable.
// Anything not explicitly allowed here is denied.
//
// MIGRATION NOTE: rules cannot hide individual fields on read. If the OLD public document still holds
//   trainees/attendance, those fields stay publicly readable until the admin signs in and the site's
//   migration (Admin > Firebase > Private-data migration) reports "verified". Deploy these rules first,
//   then sign in as admin and confirm that status.
//
// Collections
//   iti_portals/{portalId}                     PUBLIC site content (notices, videos, gallery, ...) – admin write only
//   iti_portals/{portalId}/media/*             public media – admin write only
//   iti_portals/{portalId}/comments/*          visitor comments (moderated)
//   iti_portals/{portalId}/video_comments/*    visitor video comments (moderated)
//   iti_portals/{portalId}/feedback_private/*  visitor emails – write-only for visitors, admin read
//   iti_public_results/{portalId}              PUBLIC sanitized results projection – admin write only
//   iti_private/{portalId}                     PRIVATE trainee data (reg IDs, marks, attendance) – admin ONLY
service cloud.firestore {
  match /databases/{database}/documents {

    // ---------------------------------------------------------------- helpers
    function adminEmail() { return 'dabitmunda@gmail.com'; }   // <-- REPLACE (lower-case Gmail/Google address)

    // Admin = signed in via a provider that verified the email AND exactly the configured address.
    // The email is NOT read from any database document, so no one can grant themselves admin by editing data.
    function isAdmin() {
      return request.auth != null
        && request.auth.token.email != null
        && request.auth.token.email_verified == true
        && adminEmail().matches('^[^@]+@[^@]+$')
        && request.auth.token.email.lower() == adminEmail().lower();
    }

    function portalDoc() {
      return get(/databases/$(database)/documents/iti_portals/chhotojagulia_elec_2025_27).data;
    }
    function commentsMode() {
      return portalDoc().get('settings', {}).get('commentsMode', 'moderated');
    }

    function validText(v, minLen, maxLen) {
      return v is string && v.size() >= minLen && v.size() <= maxLen;
    }

    // ------------------------------------------------- PUBLIC portal content
    match /iti_portals/{portalId} {
      allow read: if true;
      // Only the admin may write, only the allow-listed public keys, and NEVER private trainee fields.
      allow create, update: if isAdmin()
        && request.resource.data.keys().hasOnly([
             'notices','studyNotes','videos','youtubePlaylist','galleryPhotos','galleryInitialized',
             'examSchedule','internalMeta','consolidatedPublished','internalPublished','holidays',
             'settings','updatedAt'])
        && !('trainees' in request.resource.data)
        && !('attendance' in request.resource.data)
        && (!('notices' in request.resource.data) || request.resource.data.notices is list)
        && (!('studyNotes' in request.resource.data) || request.resource.data.studyNotes is list)
        && (!('videos' in request.resource.data) || request.resource.data.videos is list)
        && (!('galleryPhotos' in request.resource.data) || request.resource.data.galleryPhotos is list)
        && (!('examSchedule' in request.resource.data) || request.resource.data.examSchedule is list)
        && (!('holidays' in request.resource.data) || request.resource.data.holidays is map)
        && (!('consolidatedPublished' in request.resource.data) || request.resource.data.consolidatedPublished is bool)
        && (!('internalPublished' in request.resource.data) || request.resource.data.internalPublished is bool)
        && (!('settings' in request.resource.data) || (
             request.resource.data.settings is map
             && request.resource.data.settings.keys().hasOnly(['commentsMode','contactEmail'])
             && (!('commentsMode' in request.resource.data.settings)
                 || request.resource.data.settings.commentsMode in ['open','moderated','off'])
             && (!('contactEmail' in request.resource.data.settings)
                 || (request.resource.data.settings.contactEmail is string
                     && request.resource.data.settings.contactEmail.size() <= 120))));
      allow delete: if isAdmin();
    }

    match /iti_portals/{portalId}/media/{mediaId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // ------------------------------------------------- PRIVATE trainee data (admin only)
    match /iti_private/{portalId} {
      allow read: if isAdmin();
      allow create, update: if isAdmin()
        && request.resource.data.keys().hasOnly(['trainees','attendance','updatedAt'])
        && request.resource.data.trainees is list
        && request.resource.data.trainees.size() <= 300
        && request.resource.data.attendance is map;
      allow delete: if isAdmin();
    }

    // ------------------------------------------------- PUBLIC results projection (sanitized, allow-listed)
    // Written ONLY by the verified admin. Holds only what the public Results UI needs: no registration IDs,
    // no attendance, no unpublished 2nd-year marks, no internal marks unless internalPublished == true.
    // The site must build it from the private records (see projection spec). Rules cannot validate each list
    // element's inner fields, so the admin client must validate before publishing.
    match /iti_public_results/{portalId} {
      allow read: if true;
      allow create, update: if isAdmin()
        && request.resource.data.keys().hasOnly(['results','internalPublished','updatedAt'])
        && request.resource.data.results is list
        && request.resource.data.results.size() <= 300
        && request.resource.data.internalPublished is bool;
      allow delete: if isAdmin();
    }

    // ------------------------------------------------- Visitor comments (no PII besides a display name)
    match /iti_portals/{portalId}/comments/{cid} {
      allow read: if resource.data.status == 'visible' || isAdmin();
      allow create: if commentsMode() != 'off'
        && request.resource.data.keys().hasOnly(['name','text','status','hasEmail','createdAt'])
        && validText(request.resource.data.name, 1, 40)
        && validText(request.resource.data.text, 5, 600)
        && request.resource.data.hasEmail is bool
        && request.resource.data.createdAt == request.time
        && request.resource.data.status == (commentsMode() == 'open' ? 'visible' : 'pending');
      allow update, delete: if isAdmin();
    }

    match /iti_portals/{portalId}/video_comments/{cid} {
      allow read: if resource.data.status == 'visible' || isAdmin();
      allow create: if commentsMode() != 'off'
        && request.resource.data.keys().hasOnly(['name','text','status','videoId','createdAt'])
        && validText(request.resource.data.name, 1, 40)
        && validText(request.resource.data.text, 5, 600)
        && validText(request.resource.data.videoId, 1, 120)
        && request.resource.data.createdAt == request.time
        && request.resource.data.status == (commentsMode() == 'open' ? 'visible' : 'pending');
      allow update, delete: if isAdmin();
    }

    // Visitor emails: write-only for visitors, readable by admin only
    match /iti_portals/{portalId}/feedback_private/{cid} {
      allow create: if commentsMode() != 'off'
        && request.resource.data.keys().hasOnly(['name','email','createdAt'])
        && validText(request.resource.data.email, 0, 120)
        && request.resource.data.createdAt == request.time;
      allow read, update, delete: if isAdmin();
    }

    // Everything else: denied (implicit), stated explicitly for clarity.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
