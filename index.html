Firestore Security Rules 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function portal() { return get(/databases/$(database)/documents/iti_portals/chhotojagulia_elec_2025_27).data; }
    function mode() { return portal().get('settings', {}).get('commentsMode', 'moderated'); }
    function adminEmail() { return portal().get('settings', {}).get('adminEmail', ''); }
    function isAdmin() {
      return request.auth != null && request.auth.token.email != null
        && adminEmail() != ''
        && adminEmail().lower() == request.auth.token.email.lower();
    }
    match /iti_portals/{portalId} {
      allow read: if true;
      allow write: if isAdmin() || (adminEmail() == '' && request.auth != null);
    }
    match /iti_portals/{portalId}/media/{mediaId} {
      allow read: if true;
      allow write: if isAdmin() || (adminEmail() == '' && request.auth != null);
    }
    // Public comments (no email stored here)
    match /iti_portals/{portalId}/comments/{cid} {
      allow read: if resource.data.status == 'visible' || isAdmin();
      allow create: if mode() != 'off'
        && request.resource.data.keys().hasOnly(['name','text','status','hasEmail','createdAt'])
        && request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 40
        && request.resource.data.text is string && request.resource.data.text.size() >= 5 && request.resource.data.text.size() <= 600
        && request.resource.data.hasEmail is bool
        && request.resource.data.createdAt == request.time
        && request.resource.data.status == (mode() == 'open' ? 'visible' : 'pending');
      allow update, delete: if isAdmin();
    }
    // Per-video comments (shown under the playing video)
    match /iti_portals/{portalId}/video_comments/{cid} {
      allow read: if resource.data.status == 'visible' || isAdmin();
      allow create: if mode() != 'off'
        && request.resource.data.keys().hasOnly(['name','text','status','videoId','createdAt'])
        && request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 40
        && request.resource.data.text is string && request.resource.data.text.size() >= 5 && request.resource.data.text.size() <= 600
        && request.resource.data.videoId is string && request.resource.data.videoId.size() > 0 && request.resource.data.videoId.size() <= 120
        && request.resource.data.createdAt == request.time
        && request.resource.data.status == (mode() == 'open' ? 'visible' : 'pending');
      allow update, delete: if isAdmin();
    }
    // Private visitor emails: write-only for visitors, readable by admin only
    match /iti_portals/{portalId}/feedback_private/{cid} {
      allow create: if mode() != 'off'
        && request.resource.data.keys().hasOnly(['name','email','createdAt'])
        && request.resource.data.email is string && request.resource.data.email.size() <= 120
        && request.resource.data.createdAt == request.time;
      allow read, update, delete: if isAdmin();
    }
  }
}
